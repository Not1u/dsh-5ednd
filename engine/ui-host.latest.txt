    unreg.push(harness.handle('sheet.patch', async ({ json }) => {
      try {
        const spec = JSON.parse(json)
        const safe = String(spec.id).replace(/[^A-Za-z0-9._-]/g, '')
        const pc = overlay.get(String(spec.id)) ? overlay.get(String(spec.id)) : JSON.parse(await fsRead(CHAR_DIR + '/' + safe + '.json'))
        pc.hp = pc.hp || { current: 1, max: 1, temp: 0 }
        const p = spec.patch || {}
        const notes = []
        if (p.hpMax != null && p.hpMax !== '') { pc.hp.max = Math.max(1, Number(p.hpMax)); if (Number(pc.hp.current) > pc.hp.max) pc.hp.current = pc.hp.max; notes.push('上限HP ' + pc.hp.max) }
        if (p.hpCur != null && p.hpCur !== '') { pc.hp.current = Math.max(0, Number(p.hpCur)); if (pc.hp.current > pc.hp.max) pc.hp.max = pc.hp.current; notes.push('当前HP ' + pc.hp.current) }
        if (p.hpTemp != null && p.hpTemp !== '') { pc.hp.temp = Math.max(0, Number(p.hpTemp)); notes.push('临时HP ' + pc.hp.temp) }
        if (p.xp != null && p.xp !== '') { pc.xp = Math.max(0, Number(p.xp)); notes.push('经验 ' + pc.xp) }
        if (p.ac != null && p.ac !== '') { pc.acOverride = Math.max(1, Number(p.ac)); notes.push('AC ' + pc.acOverride) }
        pc.meta = pc.meta || {}
        pc.meta.updatedAt = nowIso()
        const r = await persist(pc)
        return { ok: true, id: pc.id, saved: !!r.ok, saveError: r.error || '', summary: (notes.length ? '已手动调整：' + notes.join('、') : '未更改任何数值'), character: JSON.stringify(pc) }
      } catch (e) { return { ok: false, error: String(e && e.message || e) } }
    }))
    unreg.push(harness.handle('multiclass.add', async ({ json }) => {
      try {
        const srd = await loadSrd()
        const spec = JSON.parse(json)
        const safe = String(spec.id).replace(/[^A-Za-z0-9._-]/g, '')
        const pc = JSON.parse(await fsRead(CHAR_DIR + '/' + safe + '.json'))
        const clsList = pc.classes || []
        const key = spec.cls
        if (!srd.classes[key]) return { ok: false, error: '职业无效' }
        if (clsList.some(c => c.class === key)) return { ok: false, error: '已拥有该职业' }
        const total = clsList.reduce((a, c) => a + (Number(c.level) || 0), 0)
        if (total + 1 > 20) return { ok: false, error: '总等级不能超过 20' }
        for (const c of clsList.concat([{ class: key }])) { const a = PRIMARY[c.class]; if (a && modOf(Number(pc.abilities[a])) < 3) return { ok: false, error: '兼职需 ' + ABF[a] + ' ≥ 13（当前不满足）' } }
        const cd = srd.classes[key]
        const conMod = modOf(Number(pc.abilities.con))
        const hpGain = (Number(cd.levelUpHitDieAverage) || 5) + conMod
        clsList.push({ class: key, level: 1 })
        pc.classes = clsList
        pc.level = total + 1
        pc.hp.max = Number(pc.hp.max) + hpGain
        pc.hp.current = Number(pc.hp.current) + hpGain
        if (cd.casting) pc.spellcasting = pc.spellcasting || { ability: cd.casting.ability, cantrips: [], spells: [] }
        pc.meta.updatedAt = nowIso()
        const r1 = await persist(pc)
        return { ok: true, id: pc.id, cls: key, level: pc.level, hpGain, saved: !!r1.ok, saveError: r1.error || '', summary: '兼职：新增 ' + cd.name + ' Lv1（总等级 ' + pc.level + '，HP +' + hpGain + '）', character: JSON.stringify(pc) }
      } catch (e) { return { ok: false, error: String(e && e.message || e) } }
    }))
    return () => { unreg.forEach(f => f()) }
  }
}
