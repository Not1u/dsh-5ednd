function LevelSet(props) {
  const s = props.s
  const [target, setTarget] = React.useState(Number(s.level) || 1)
  const [info, setInfo] = React.useState(null)
  const [states, setStates] = React.useState([])
  const [busy, setBusy] = React.useState(false)
  const [msg, setMsg] = React.useState('')
  const calc = () => { call('levelset.info', { id: s.id, target: target }).then(r => { if (r && r.ok) { setInfo(r); setStates((r.pending || []).map(() => ({}))) } else setMsg((r && r.error) || '计算失败') }) }
  const doApply = async () => {
    setBusy(true); setMsg('')
    try {
      const choices = (info.pending || []).map((ch, i) => { const st = states[i] || {}; if (ch.kind === 'subclass') return { kind: 'subclass', key: st.key, picks: st.picks }; if (ch.kind === 'pactBoon') return { kind: 'pactBoon', key: st.key }; if (ch.kind === 'invocation') return { kind: 'invocation', keys: st.keys }; if (ch.kind === 'asi') return Object.assign({ kind: 'asi' }, st); if (ch.kind === 'cantrip' || ch.kind === 'spell') return { kind: ch.kind, keys: st.keys }; return null }).filter(x => x && (x.key || x.keys || x.mode))
      const r = await call('levelset.apply', { json: JSON.stringify({ id: s.id, target: target, choices: choices }) })
      if (r && r.ok) { setMsg(r.summary + (r.saved ? '（已保存）' : '（未保存：' + (r.saveError || '未知原因') + '）')); setInfo(null); if (props.onReload) props.onReload() } else setMsg((r && r.error) || '应用失败')
    } catch (e) { setMsg(String(e && e.message || e)) }
    setBusy(false)
  }
  const kids = []
  kids.push(h('div', { key: 'h', style: { display: 'flex', alignItems: 'center', gap: 8 } }, h('b', null, '等级调整'), h('div', { className: 'dndp-sp' }), h('button', { className: 'dndp-ico', onClick: () => setTarget(t => Math.max(1, t - 1)) }, '−'), h('b', null, 'Lv' + target), h('button', { className: 'dndp-ico', onClick: () => setTarget(t => Math.min(20, t + 1)) }, '+'), h('button', { className: 'dndp-btn', onClick: calc }, '计算升级内容')))
  if (info && info.ok) {
    if (info.same) kids.push(h('div', { key: 's', className: 'dndp-note' }, '当前已是 Lv' + info.level))
    else {
      kids.push(h('div', { key: 'd', className: 'dndp-note' }, 'Lv' + info.level + ' → Lv' + info.target + '：每级 HP ' + (info.hpPerLevel >= 0 ? '+' : '') + info.hpPerLevel))
      if (info.slotsAfter && info.slotsAfter.length) kids.push(h('div', { key: 'sl', className: 'dndp-note' }, '调整后法术位：' + info.slotsAfter.map(x => x.lv + '环×' + x.max).join('、')))
      ;(info.pending || []).forEach((ch, i) => kids.push(h('div', { key: 'p' + i, style: { marginTop: 8 } }, h(ChoiceBox, { ch: ch, st: states[i] || {}, setSt: v => setStates(arr => { const n = arr.slice(); n[i] = v; return n }) }))))
      if (!info.pending || !info.pending.length) kids.push(h('div', { key: 'n', className: 'dndp-note' }, '该等级区间无需额外选择'))
      kids.push(h('div', { key: 'go', style: { marginTop: 10 } }, h('button', { className: 'dndp-btn primary', disabled: busy, onClick: doApply }, '应用等级调整')))
    }
  }
  if (msg) kids.push(h('div', { key: 'm', className: 'dndp-note hl' }, msg))
  return h('div', { className: 'dndp-panel' }, kids)
}

function RulesPanel(props) {
  const [q, setQ] = React.useState('')
  const [book, setBook] = React.useState('')
  const [items, setItems] = React.useState(null)
  const [stats, setStats] = React.useState(null)
  const [openId, setOpenId] = React.useState('')
  const [text, setText] = React.useState('')
  const [busy, setBusy] = React.useState(false)
  const [msg, setMsg] = React.useState('')
  const [recent, setRecent] = React.useState(() => { try { const v = localStorage.getItem('dnd5e.recentRules'); return v ? JSON.parse(v) : [] } catch (e) { return [] } })
  React.useEffect(() => {
    call('rules.stats', {}).then(r => { if (r && r.ok) setStats(r); else setMsg('规则索引不可用：' + ((r && r.error) || '') + '（在仓库运行 node tools/index-rules.mjs 生成索引）') }).catch(e => setMsg(String(e && e.message || e)))
  }, [])
  const remember = (term) => {
    const next = [term].concat(recent.filter(x => x !== term)).slice(0, 6)
    setRecent(next)
    try { localStorage.setItem('dnd5e.recentRules', JSON.stringify(next)) } catch (e) { }
  }
  const terms = () => q.trim().split(/[\s,，]+/).filter(Boolean)
  const run = async (override) => {
    const query = typeof override === 'string' ? override : q
    if (!query.trim()) return
    if (override) setQ(query)
    setBusy(true); setMsg(''); setItems(null); setOpenId(''); setText('')
    try {
      const args = { query: query, limit: 12 }
      if (book) args.book = book
      const r = await call('rules.search', args)
      if (r && r.ok) { setItems(r.items || []); remember(query.trim()); if (!(r.items || []).length) setMsg('未命中：' + query) } else setMsg((r && r.error) || '检索失败')
    } catch (e) { setMsg(String(e && e.message || e)) }
    setBusy(false)
  }
  const open = async (id) => {
    if (openId === id) { setOpenId(''); return }
    setOpenId(id); setText('读取中…')
    try {
      const r = await call('rules.read', { id: id })
      setText(r && r.ok ? r.entry.text : ('读取失败：' + ((r && r.error) || '')))
    } catch (e) { setText(String(e && e.message || e)) }
  }
  const hl = (str) => {
    const ts = terms().filter(t => t.length > 0)
    if (!ts.length) return str
    const out = []
    let rest = String(str)
    let guard = 0
    while (rest.length && guard++ < 200) {
      const lower = rest.toLowerCase()
      let best = -1, bestLen = 0
      for (const t of ts) { const i = lower.indexOf(t.toLowerCase()); if (i >= 0 && (best < 0 || i < best)) { best = i; bestLen = t.length } }
      if (best < 0) { out.push(rest); break }
      if (best > 0) out.push(rest.slice(0, best))
      out.push(h('span', { key: 'm' + guard, className: 'dndp-hl' }, rest.slice(best, best + bestLen)))
      rest = rest.slice(best + bestLen)
    }
    return out
  }
  const kids = []
  kids.push(h('div', { key: 'h', style: { display: 'flex', alignItems: 'center', gap: 8 } },
    h('b', null, '规则查找'),
    h('div', { className: 'dndp-sp' }),
    stats ? h('span', { className: 'dndp-chip' }, (stats.total || 0) + ' 条 / ' + ((stats.books || []).length) + ' 本书') : null))
  kids.push(h('div', { key: 'f', style: { display: 'flex', gap: 6, marginTop: 8 } },
    h('input', { key: 'i', className: 'dndp-input', placeholder: '关键词，空格=同时满足，如：擒抱 体型 / 火球术 伤害', value: q, onChange: e => setQ(e.target.value), onKeyDown: e => { if (e.key === 'Enter') run() } }),
    h('button', { key: 'b', className: 'dndp-btn primary', disabled: busy, onClick: () => run() }, '检索')))
  if (recent.length) kids.push(h('div', { key: 'rec', className: 'dndp-tags', style: { marginTop: 6 } },
    h('span', { key: 'lb', className: 'dndp-note' }, '最近：'),
    recent.map((r2, i) => h('button', { key: i, className: 'dndp-tag', style: { cursor: 'pointer' }, onClick: () => run(r2) }, r2))))
  if (stats && (stats.books || []).length) {
    const chips = [h('button', { key: 'all', className: 'dndp-tag', style: { cursor: 'pointer', borderColor: book === '' ? 'var(--dsw-alias-brand-primary)' : undefined }, onClick: () => setBook('') }, '全部')]
    ;(stats.books || []).forEach((b, i) => chips.push(h('button', { key: i, className: 'dndp-tag', style: { cursor: 'pointer', borderColor: book === b.book ? 'var(--dsw-alias-brand-primary)' : undefined }, onClick: () => setBook(book === b.book ? '' : b.book) }, b.book + ' ' + b.entries)))
    kids.push(h('div', { key: 'bk', className: 'dndp-tags', style: { marginTop: 6 } }, chips))
  }
  if (items && items.length) {
    kids.push(h('div', { key: 'l', className: 'dndp-label' }, '结果 ' + items.length + ' 条（点击展开全文）'))
    items.forEach((it, i) => {
      kids.push(h('div', { key: 'r' + i, className: 'dndp-sect', style: { marginTop: 6 } },
        h('button', { className: 'dndp-sec', onClick: () => open(it.id) },
          h('span', { className: 'dndp-chev' + (openId === it.id ? ' open' : '') }, '▶'),
          h('span', null, hl(it.title)),
          h('div', { className: 'dndp-sp' }),
          h('span', { className: 'dndp-chip' }, it.book)),
        openId === it.id
          ? h('div', { className: 'dndp-body' }, h('div', { style: { whiteSpace: 'pre-wrap', fontSize: 12, maxHeight: 340, overflow: 'auto' } }, text))
          : h('div', { className: 'dndp-body' }, h('div', { className: 'dndp-note' }, hl(it.snippet)))))
    })
  }
  if (msg) kids.push(h('div', { key: 'm', className: 'dndp-note hl' }, msg))
  return h('div', { className: 'dndp-panel' }, kids)
}

function DiceBar(props) {

const cellName = (x, y) => {
  let s = '', n = Math.max(0, Math.floor(x))
  do { s = String.fromCharCode(65 + (n % 26)) + s; n = Math.floor(n / 26) - 1 } while (n >= 0)
  return s + (Math.floor(y) + 1)
}
function cellDist(a, b) {
  const dx = Math.abs(a.x - b.x), dy = Math.abs(a.y - b.y)
  const mx = Math.max(dx, dy), mn = Math.min(dx, dy)
  return { cells: mx, feetSimple: mx * 5, feetStandard: (mx + Math.floor(mn / 2)) * 5 }
}

    const x = Math.floor((ev.clientX - r.left) / tile)
    const y = Math.floor((ev.clientY - r.top) / tile)
    if (!map || x < 0 || y < 0 || x >= map.w || y >= map.h) return null
    return { x, y }
  }
  const moveToken = async (tk, c) => {
    const d = cellDist({ x: tk.x, y: tk.y }, c)
    const over = tk.speed ? d.feetStandard - tk.speed : 0
    setMap(m => Object.assign({}, m, { tokens: (m.tokens || []).map(t => t.id === tk.id ? Object.assign({}, t, { x: c.x, y: c.y }) : t) }))
    setMsg(tk.name + ' ' + cellName(tk.x, tk.y) + ' → ' + cellName(c.x, c.y) + '：' + d.feetStandard + ' 尺（' + d.cells + ' 格）' + (over > 0 ? ' ⚠ 超出速度 ' + tk.speed + ' 尺，超 ' + over + ' 尺' : '') + '，已记入战斗记录')
    const r = await call('map.token.move', { id: tk.id, x: c.x, y: c.y })
    if (!r || !r.ok) setErr((r && r.error) || '移动失败')
  }
  const onCellClick = (c) => {
    if (tool === 'measure') {
      if (!meas.a || (meas.a && meas.b)) setMeas({ a: c, b: null })
      else setMeas({ a: meas.a, b: c })
      return
    }
    if (sel) {
      const tk = (map.tokens || []).find(t => t.id === sel)
      if (tk && (tk.x !== c.x || tk.y !== c.y)) moveToken(tk, c)
      return
    }
    setMsg('先点一个单位图标选中它（或用拖动直接移动），再点目标格 = 走过去。')
  }
  const onTokDown = (ev, tk) => {
    if (tool !== 'move') { setSel(tk.id); return }
    ev.preventDefault()
    setSel(tk.id)
    setDrag({ id: tk.id, from: { x: tk.x, y: tk.y }, cur: { x: tk.x, y: tk.y } })
  }
  React.useEffect(() => {
    if (!drag) return undefined
    const onMove = (ev) => { const c = cellFromEvent(ev); if (c) setDrag(d => d ? Object.assign({}, d, { cur: c }) : d) }
    const onUp = (ev) => {
      const c = cellFromEvent(ev)
      const tk = (map.tokens || []).find(t => t.id === drag.id)
      if (tk && c && (c.x !== drag.from.x || c.y !== drag.from.y)) moveToken(tk, c)
      setDrag(null)
    }
    window.addEventListener('pointermove', onMove)
    window.addEventListener('pointerup', onUp)
    return () => { window.removeEventListener('pointermove', onMove); window.removeEventListener('pointerup', onUp) }
  }, [drag, map, tile])
  if (!map) return h('div', { className: 'dndp-root' }, h('div', { className: 'dndp-header' }, h('span', { className: 'dndp-ic' }, '🗺'), h('span', null, '战术地图'), h('div', { className: 'dndp-sp' }), err ? h('span', { className: 'dndp-chip' }, err) : h('span', { className: 'dndp-chip' }, '加载中…')))
  const tokens = (map.tokens || []).filter(t => !t.hidden)
  const cells = []
  for (let y = 0; y < map.h; y++) {
    const row = (map.terrain[y] || '.'.repeat(map.w)).split('')
    for (let x = 0; x < map.w; x++) {
      const ch = row[x] || '.'
      const td = TERRAIN[ch] || TERRAIN['.']
      cells.push(h('div', { key: x + '-' + y, className: 'dndp-cell g-' + td.k, onClick: () => onCellClick({ x, y }), title: cellName(x, y) + ' · ' + td.n + (td.cover ? '（AC+2）' : '') }))
    }
  }
  const m1 = meas.a, m2 = meas.b
  const md = (m1 && m2) ? cellDist(m1, m2) : null
  const grid = h('div', { ref: gridRef, className: 'dndp-mapgrid', style: { gridTemplateColumns: 'repeat(' + map.w + ', var(--dndp-tile))', '--dndp-tile': tile + 'px', position: 'relative' } },
    cells,
    tokens.map(tk => {
      const size = (Number(tk.size) || 1)
      const isDrag = drag && drag.id === tk.id
      const px = isDrag ? drag.cur.x : tk.x, py = isDrag ? drag.cur.y : tk.y
      const bar = (tk.hp != null && tk.max) ? (() => { const pct = Math.max(0, Math.min(100, Math.round(tk.hp / tk.max * 100))); return h('div', { className: 'dndp-tokbar' + (pct <= 25 ? ' crit' : (pct <= 50 ? ' low' : '')) }, h('i', { style: { width: pct + '%' } })) })() : null
      const d = isDrag ? cellDist(drag.from, drag.cur) : null
      return h('div', {
        key: tk.id,
        className: 'dndp-tok k-' + (tk.kind || 'enemy') + (isDrag ? ' drag' : '') + (sel === tk.id ? ' sel' : ''),
        style: { left: px * tile + 2, top: py * tile + 2, width: size * tile - 4, height: size * tile - 4, background: tk.color || undefined, zIndex: isDrag ? 20 : 5 },
        onPointerDown: e => onTokDown(e, tk),
        title: tk.name + ' @ ' + cellName(px, py) + (tk.hp != null ? ('｜HP ' + tk.hp + (tk.max ? '/' + tk.max : '')) : '') + (tk.speed ? ('｜速度 ' + tk.speed + ' 尺') : '') + (tk.note ? ('｜' + tk.note) : '') + (isDrag ? ('｜正在移动 ' + d.feetStandard + ' 尺') : ''),
      }, h('span', { className: 'tkname' }, tk.name.slice(0, 4)), bar)
    }),
    (md && m1 && m2) ? h('svg', { className: 'dndp-measure', width: '100%', height: '100%' },
      h('line', { x1: (m1.x + 0.5) * tile, y1: (m1.y + 0.5) * tile, x2: (m2.x + 0.5) * tile, y2: (m2.y + 0.5) * tile, stroke: 'var(--dsw-alias-brand-primary)', strokeWidth: 2, strokeDasharray: '5 4' }),
      h('circle', { cx: (m1.x + 0.5) * tile, cy: (m1.y + 0.5) * tile, r: 4, fill: 'var(--dsw-alias-brand-primary)' }),
      h('circle', { cx: (m2.x + 0.5) * tile, cy: (m2.y + 0.5) * tile, r: 4, fill: 'var(--dsw-alias-brand-primary)' })) : null,
    (md && m1 && m2) ? h('div', { className: 'dndp-mlabel', style: { left: ((m1.x + m2.x) / 2 + 0.5) * tile, top: ((m1.y + m2.y) / 2 + 0.5) * tile } }, cellName(m1.x, m1.y) + '→' + cellName(m2.x, m2.y) + '：' + md.feetStandard + ' 尺 / ' + md.cells + ' 格') : null,
    (sel && map.tokens.find(t => t.id === sel)) ? (() => { const tk = map.tokens.find(t => t.id === sel); const reach = tk.speed ? Math.floor(tk.speed / 5) : 6; const ring = []; for (let dy = -reach; dy <= reach; dy++) for (let dx = -reach; dx <= reach; dx++) { const c = { x: tk.x + dx, y: tk.y + dy }; if (c.x < 0 || c.y < 0 || c.x >= map.w || c.y >= map.h) continue; const dd = cellDist({ x: tk.x, y: tk.y }, c); if (dd.feetStandard > 0 && dd.feetStandard <= tk.speed) ring.push(h('div', { key: 'r' + dx + '_' + dy, style: { position: 'absolute', left: c.x * tile, top: c.y * tile, width: tile, height: tile, background: 'rgba(90,160,255,.16)', border: '1px solid rgba(90,160,255,.25)', pointerEvents: 'none', zIndex: 4 } })) } return ring })() : null)
  const tools = h('div', { className: 'dndp-maptools' },
    h('span', { className: 'dndp-chip' }, map.name || '未命名地图'),
    h('span', { className: 'dndp-chip' }, map.w + '×' + map.h + ' 格（每格 5 尺）'),
    h('button', { className: 'dndp-btn tiny' + (tool === 'move' ? ' primary' : ''), onClick: () => setTool('move') }, '移动'),
    h('button', { className: 'dndp-btn tiny' + (tool === 'measure' ? ' primary' : ''), onClick: () => { setTool('measure'); setMeas({ a: null, b: null }) } }, '测距'),
    h('button', { className: 'dndp-btn tiny', onClick: () => { setMeas({ a: null, b: null }); setSel(null); setMsg('') } }, '清除选点'),
    h('span', { className: 'dndp-note' }, '格距'),
    h('input', { type: 'range', min: 16, max: 60, value: tile, onChange: e => setTile(Number(e.target.value)), style: { width: 92 } }),
    h('span', { className: 'dndp-note' }, tile + 'px'),
    h('button', { className: 'dndp-btn tiny', onClick: load }, '刷新'),
    h('button', { className: 'dndp-btn tiny', onClick: async () => { setMsg('导出中…'); const r = await call('ext.image.map', {}); setMsg(r && r.ok ? r.summary : ('导出失败：' + ((r && r.error) || '需要重启 DSH 以加载 plugins/')) ) } }, '⬇ 导出图片'),
    props.onToggleFull ? h('button', { className: 'dndp-btn tiny', onClick: props.onToggleFull }, props.full ? '退出全屏' : '⤢ 全屏') : null)
  const legend = h('div', { className: 'dndp-legend' }, h('span', null, '图例：'), (map.legend || Object.keys(TERRAIN).map(k => [k, TERRAIN[k].n])).map((L, i) => h('span', { key: i }, h('span', { className: 'sw g-' + ((TERRAIN[L[0]] || {}).k || 'void') }), L[1])))
  const body = [tools, h('div', { key: 'm', style: { padding: '0 10px 10px' } }, h('div', { className: 'dndp-mapwrap', style: { maxHeight: props.full ? 'none' : 420 } }, grid)), msg ? h('div', { key: 'g', className: 'dndp-note hl', style: { padding: '0 12px' } }, msg) : null, err ? h('div', { key: 'e', className: 'dndp-err' }, err) : null, legend,
    h('div', { key: 't', className: 'dndp-note', style: { padding: '0 12px 12px' } }, '用法：拖动图标移动单位（自动按 5 尺格换算距离并写入战斗记录）；点「测距」后点两格看距离；选中单位会显示其速度范围（蓝色水波纹区）。')]
  if (props.full) return h('div', { className: 'dndp-overlay' }, h('div', { className: 'dndp-ovhead' }, h('span', { className: 'dndp-ic' }, '🗺'), h('span', null, '战术地图 — ' + (map.name || '未命名')), h('div', { className: 'dndp-sp' }), h('button', { className: 'dndp-btn', onClick: props.onToggleFull }, '关闭（Esc）')), h('div', { className: 'dndp-ovbody' }, body))
  return h('div', { className: 'dndp-root' }, h('div', { className: 'dndp-header' }, h('span', { className: 'dndp-ic' }, '🗺'), h('span', null, '战术地图'), h('div', { className: 'dndp-sp' }), h('button', { className: 'dndp-btn', onClick: props.onBack }, '返回')), h('div', { className: 'dndp-scroll' }, body))
}
function PartyView() {
  const [mode, setMode] = React.useState('list')
  const [list, setList] = React.useState(null)
  const [err, setErr] = React.useState('')
  const [busy, setBusy] = React.useState(false)
  const [sel, setSel] = React.useState(null)
  const [loading, setLoading] = React.useState(false)
  const [mapFull, setMapFull] = React.useState(false)
  function refresh() { setBusy(true); setErr(''); call('party.list', {}).then(r => { if (r && r.ok) setList(r.items); else setErr((r && r.error) || 'list failed') }).catch(e => setErr(String(e && e.message || e))).then(() => setBusy(false)) }
  function open(id) { setLoading(true); setErr(''); call('party.sheet', { id }).then(r => { if (r && r.ok) setSel(r.sheet); else setErr((r && r.error) || 'sheet failed') }).catch(e => setErr(String(e && e.message || e))).then(() => setLoading(false)) }
  React.useEffect(() => { refresh() }, [])
  React.useEffect(() => {
    if (!mapFull) return undefined
    const onKey = (e) => { if (e.key === 'Escape') setMapFull(false) }
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [mapFull])
  if (mapFull) return h(Boundary, null, h(MapPanel, { full: true, onToggleFull: () => setMapFull(false) }))
  if (mode === 'build') return h(Boundary, null, h(Wizard, { onDone: () => { setMode('list'); refresh() }, onBack: () => setMode('list') }))
  if (mode === 'log') return h(Boundary, null, h(LogView, { party: list || [], onBack: () => { setMode('list'); refresh() } }))
  if (mode === 'map') return h(Boundary, null, h(MapPanel, { onBack: () => { setMode('list'); refresh() }, onToggleFull: () => setMapFull(true) }))
  if (sel) return h('div', { className: 'dndp-root' }, h('div', { className: 'dndp-header' }, h('span', { className: 'dndp-ic' }, '⚔'), h('span', null, sel.name), h('div', { className: 'dndp-sp' }), h(XpWidget, { s: sel }), h('button', { className: 'dndp-btn', onClick: () => setMode('map') }, '🗺 地图'), h('button', { className: 'dndp-btn', onClick: () => setMode('log') }, '🎲 战斗记录'), h('button', { className: 'dndp-btn', onClick: () => setSel(null) }, '返回列表')), h('div', { className: 'dndp-scroll' }, loading ? h('div', { className: 'dndp-empty' }, '加载中…') : h(Boundary, null, h(Detail, { s: sel, onReload: () => { open(sel.id); refresh() } }))))
  const rows = []
  if (busy || loading) rows.push(h('div', { key: 'b', className: 'dndp-empty' }, '加载中…'))
  if (err) rows.push(h('div', { key: 'e', className: 'dndp-err' }, err))
  if (list && !list.length) rows.push(h('div', { key: 'z', className: 'dndp-empty' }, '暂无角色，点“+ 新建角色”'))
  ;(list || []).forEach((l, i) => rows.push(h('div', { key: i, className: 'dndp-row', onClick: () => open(l.id) }, h('div', { className: 'dndp-ava' }, (l.name || '?').slice(0, 1)), h('div', { className: 'dndp-main' }, h('div', { className: 'dndp-name' }, l.name), h('div', { className: 'dndp-sub' }, (l.race || '') + (l.cls ? ' · ' + l.cls : ''))), h('span', { className: 'dndp-chip' }, 'L' + l.level), h('span', { className: 'dndp-chip' }, 'HP ' + l.hp + '/' + l.max))))
  return h('div', { className: 'dndp-root' }, h('div', { className: 'dndp-header' }, h('span', { className: 'dndp-ic' }, '⚔'), h('span', null, 'D&D 5e 角色'), h('div', { className: 'dndp-sp' }), h(UiBadge, null), h('button', { className: 'dndp-btn', onClick: () => setMode('map') }, '🗺 地图'), h('button', { className: 'dndp-btn', onClick: () => setMode('log') }, '🎲 战斗记录'), h('button', { className: 'dndp-btn primary', onClick: () => setMode('build') }, '+ 新建角色')), h('div', { className: 'dndp-scroll' }, rows))
}

      let d = null
      if (bsb && typeof bsb.registerTab === 'function') d = bsb.registerTab({ id: 'dnd5e:party', title: 'D&D 角色', single: true, component: () => h(PartyView, null) })
